Here is all the documentation for assignment 5! For this assignment, I expanded on the libraries assignment. This time, you enter a pin to make the fan connected to the DC motor briefly speed up, and then slow down. I used a N-MOSFET transistor to control the voltage for the pin. The 3-6 volt DC motor gives 780mA at maximum efficiency, which is much more than the pin's 40mA. The transistor helps control the current without damaging the pin or Arduino. I also used a flyback diode to help with voltage spikes.

This photo shows the circuit. There is a 4x4 membrane keypad, a dc motor, a transistor, a flyback diode, a breadboard, and an arduino board.

Arduino Code

        #include  // adds keypad library

        const byte ROWS = 4; // sets row number
        const byte COLS = 4; // sets columns number

        int motorPin = 3; // sets PWM pin

        // sets up the rows/cols as seen on the physical keypad
        char keys[ROWS][COLS] = {
          {'1','2','3','4'},
          {'4','5','6','B'},
          {'7','8','9','C'},
          {'*','0','#','D'}
        };

        byte rowPins[ROWS] = {11, 10, 9, 8}; // set the pins to rows
        byte colPins[COLS] = {7, 6, 5, 4}; // set the pins to cols

        Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS); // use a keypad library function to set up keypad

        String code = "2225"; // the correct code
        String entered = ""; // the entered code


        void setup() {
          Serial.begin(9600); // starts serial
          pinMode(motorPin, OUTPUT); // sets up motor - pin state begins as LOW/0 and is an input before pinMode setup
        }

        void loop() {
          char key = keypad.getKey(); // get the entered key

          if (key) {
            Serial.println(key);

            if (key == '#') { // user pressed enter
              if (code == entered) { // check the code was correct
                activateMotor(); // function to move motor
              } else {
                // really only for me to check
                Serial.println("incorrect");
              }
              entered = ""; // resets code
            } else if (key == '*') { // user wants to reset code
                entered = "";
            } else {
              entered += key;
            }
          }
        }

        // makes the motor fan move to show code was correct
        void activateMotor() {
          Serial.println("activated");
          // Speed up
          for (int i = 0; i <= 255; i++) {
            analogWrite(motorPin, i);
            delay(10);
          }
          // Slow down
          for (int i = 255; i >= 0; i--) {
            analogWrite(motorPin, i);
            delay(10);
          }
        }
